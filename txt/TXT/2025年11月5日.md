以下是脉搏折线图的实现代码，基于提供的数据和容器要求，重点展示脉搏随时间的变化趋势：

```vue
<template>
  <div class="data-char" v-if="actionModel === 'char'" id="char-container"></div>
</template>

<script>
import * as echarts from 'echarts';

export default {
  data() {
    return {
      actionModel: 'char', // 控制图表显示的状态
      pulseData: {
        "00:45": ["96"],
        "01:35": ["95"],
        // 可添加更多数据
      }
    };
  },
  mounted() {
    // 页面加载完成后初始化图表
    this.renderPulseChart();
  },
  methods: {
    renderPulseChart() {
      // 1. 获取容器并初始化图表
      const chartDom = document.getElementById('char-container');
      if (!chartDom) return; // 防止容器不存在时报错
      
      const myChart = echarts.init(chartDom);
      
      // 2. 处理数据：按时间排序
      const timeList = Object.keys(this.pulseData);
      // 按时间升序排序（时:分格式）
      timeList.sort((a, b) => {
        const [hourA, minuteA] = a.split(':').map(Number);
        const [hourB, minuteB] = b.split(':').map(Number);
        return (hourA * 60 + minuteA) - (hourB * 60 + minuteB);
      });
      
      // 提取脉搏数据（转换为数字类型）
      const pulseValues = timeList.map(time => Number(this.pulseData[time][0]));
      
      // 3. 图表配置
      const option = {
        title: {
          text: '脉搏监测趋势',
          left: 'center',
          textStyle: {
            fontSize: 16
          }
        },
        tooltip: {
          trigger: 'axis',
          formatter: function(params) {
            // 自定义提示框内容
            return `
              <div>时间：${params[0].name}</div>
              <div>脉搏：${params[0].value} 次/分</div>
            `;
          }
        },
        grid: {
          left: '5%',
          right: '5%',
          bottom: '15%',
          top: '15%',
          containLabel: true
        },
        xAxis: {
          type: 'category',
          data: timeList,
          axisLabel: {
            rotate: 45, // 标签旋转避免重叠
            interval: 0, // 显示所有标签
            fontSize: 12
          },
          name: '时间',
          nameTextStyle: {
            fontSize: 14,
            padding: [10, 0, 0, 0] // 调整名称位置
          }
        },
        yAxis: {
          type: 'value',
          name: '脉搏(次/分)',
          nameTextStyle: {
            fontSize: 14
          },
          min: 50, // 脉搏合理最小值
          max: 120, // 脉搏合理最大值
          interval: 10, // 刻度间隔
          axisLabel: {
            formatter: '{value}'
          }
        },
        series: [
          {
            name: '脉搏',
            type: 'line',
            data: pulseValues,
            symbol: 'circle', // 数据点形状
            symbolSize: 8, // 数据点大小
            lineStyle: {
              width: 3,
              color: '#4CAF50' // 脉搏线绿色（健康色）
            },
            itemStyle: {
              color: '#4CAF50'
            },
            // 显示数据点标签
            label: {
              show: true,
              position: 'top',
              fontSize: 12
            }
          }
        ]
      };
      
      // 4. 设置配置并渲染
      myChart.setOption(option);
      
      // 5. 监听窗口大小变化，自动调整图表
      window.addEventListener('resize', () => {
        myChart.resize();
      });
      
      // 6. 组件销毁时清理图表实例
      this.$once('hook:beforeDestroy', () => {
        myChart.dispose();
      });
    }
  }
};
</script>

<style scoped>
.data-char {
  width: 100%;
  height: 400px; /* 图表高度，可根据需求调整 */
  margin: 20px 0;
}
</style>
```

### 关键特性说明：
1. **数据处理**：
    - 自动对时间轴进行排序（按小时和分钟计算总分钟数排序）
    - 将脉搏数据从字符串转换为数字类型，确保图表正确渲染

2. **图表配置**：
    - 横轴（xAxis）显示时间，标签旋转45度避免重叠
    - 纵轴（yAxis）设置合理的脉搏范围（50-120次/分），符合生理常识
    - 使用绿色作为主色调（象征健康），突出脉搏数据的视觉识别性

3. **交互体验**：
    - 鼠标悬停时显示详细的时间和脉搏值
    - 数据点显示具体数值，便于快速查看
    - 支持窗口大小变化时自动调整图表尺寸

4. **样式适配**：
    - 合理设置网格边距，确保标题、坐标轴不被遮挡
    - 数据点标签位置居上，避免与线条重叠

直接使用此代码即可展示脉搏折线图，若数据量增加（如更多时间点），图表会自动适配并保持排序后的时间轴展示。
