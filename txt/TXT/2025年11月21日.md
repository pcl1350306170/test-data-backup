    // 创建新的deptIntros对象

      if (dept.children) {
        const found = findDept(dept.children, id);
        if (found) return found;
      }
    }
    return null;
  };
  
  const dept = findDept(workerInfo.deptOptions, deptId);
  return dept ? dept.label : `未知科室(${deptId})`;
};
