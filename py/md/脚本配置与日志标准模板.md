# Python 脚本配置与日志标准模板

> 适用于项目中所有带 GUI 的 `.pyw` / `.py` 脚本。  
> 按照本文档添加后，脚本即可具备：**配置自动保存/加载** + **统一日志模块**。

---

## 一、配置持久化（pathlib + JSON）

### 1.1 路径常量（放在文件顶部，import 之后）

```python
import os
import json
import sys
from pathlib import Path

SCRIPT_DIR = Path(os.path.abspath(os.path.dirname(__file__)))
SCRIPT_NAME = "脚本名"                    # 不含扩展名，如 "file_mover"
CONFIG_DIR = SCRIPT_DIR / "json"
CONFIG_PATH = CONFIG_DIR / f"config_{SCRIPT_NAME}.json"
CONFIG_DIR.mkdir(exist_ok=True)
```

**约定**：
- 配置文件统一存放在脚本同级的 `json/` 子目录
- 文件名格式：`config_{脚本名}.json`
- 目录不存在时自动创建

### 1.2 保存配置方法

```python
def _save_config(self):
    config = {
        "key1": self.var1.get(),
        "key2": self.var2.get(),
        # ... 按需添加
    }
    try:
        with open(CONFIG_PATH, "w", encoding="utf-8") as f:
            json.dump(config, f, ensure_ascii=False, indent=2)
        logger.info("配置已保存: %s" % CONFIG_PATH)
    except Exception as e:
        logger.error("保存配置失败: %s" % e)
```

### 1.3 加载配置方法

```python
def _load_config(self):
    try:
        if CONFIG_PATH.exists():
            with open(CONFIG_PATH, "r", encoding="utf-8") as f:
                cfg = json.load(f)
            if cfg.get("key1"):
                self.var1.set(cfg["key1"])
            if cfg.get("key2"):
                self.var2.set(cfg["key2"])
            # ... 按需添加
            logger.info("已加载配置: %s" % CONFIG_PATH)
    except Exception as e:
        logger.error("加载配置失败: %s" % e)
```

### 1.4 调用时机

```python
# __init__ 中：UI 创建完毕后加载
self._build_ui()
self._load_config()

# 用户点击"开始/执行"按钮时保存
def _start(self):
    self._save_config()
    # ... 业务逻辑
```

## 二、统一日志模块（log_utils）

### 2.1 导入代码（紧跟配置常量之后）

```python
# ---------- 日志（可选依赖） ----------
_PY_DIR = str(SCRIPT_DIR.parent)          # 指向 py/ 目录
if _PY_DIR not in sys.path:
    sys.path.insert(0, _PY_DIR)

try:
    from log_utils import get_logger
    logger = get_logger(SCRIPT_NAME)
except Exception:
    class _DummyLogger:
        def info(self, *a, **kw): pass
        def warning(self, *a, **kw): pass
        def error(self, *a, **kw): pass
        def debug(self, *a, **kw): pass
    logger = _DummyLogger()
```

**要点**：
- 必须先把 `py/` 目录加入 `sys.path`，否则子目录下的脚本找不到 `log_utils`
- `try/except` 兜底，确保 `log_utils` 不可用时脚本仍能正常运行
- `_DummyLogger` 提供空实现，所有 `logger.info()` 等调用静默跳过

### 2.2 日志存储结构（自动管理，无需手动配置）

```
py/logs/
    2026-09/
        脚本名_2026-09-01.log
        脚本名_2026-09-02.log
```

- 按月分目录，自动创建
- 超过 30 天的日志自动清理
- 同时输出到文件和控制台

## 三、完整模板（直接复制修改）

```python
# -*- coding: utf-8 -*-
"""脚本功能描述"""

import os
import json
import sys
import tkinter as tk
from pathlib import Path
from tkinter import ttk, filedialog, messagebox

# ================== 配置与常量 ==================
SCRIPT_DIR = Path(os.path.abspath(os.path.dirname(__file__)))
SCRIPT_NAME = "my_script_name"
CONFIG_DIR = SCRIPT_DIR / "json"
CONFIG_PATH = CONFIG_DIR / f"config_{SCRIPT_NAME}.json"
CONFIG_DIR.mkdir(exist_ok=True)

# ---------- 日志（可选依赖） ----------
_PY_DIR = str(SCRIPT_DIR.parent)
if _PY_DIR not in sys.path:
    sys.path.insert(0, _PY_DIR)

try:
    from log_utils import get_logger
    logger = get_logger(SCRIPT_NAME)
except Exception:
    class _DummyLogger:
        def info(self, *a, **kw): pass
        def warning(self, *a, **kw): pass
        def error(self, *a, **kw): pass
        def debug(self, *a, **kw): pass
    logger = _DummyLogger()


class App:
    def __init__(self, root: tk.Tk):
        self.root = root
        root.title("工具标题")
        root.geometry("700x500")

        # 变量定义
        self.input_path = tk.StringVar()
        self.output_dir = tk.StringVar()

        # UI 创建
        self._build_ui()
        # 加载上次配置
        self._load_config()

    def _build_ui(self):
        # ... 界面控件 ...
        pass

    # ---------- 配置持久化 ----------
    def _save_config(self):
        config = {
            "input_path": self.input_path.get(),
            "output_dir": self.output_dir.get(),
        }
        try:
            with open(CONFIG_PATH, "w", encoding="utf-8") as f:
                json.dump(config, f, ensure_ascii=False, indent=2)
            logger.info("配置已保存: %s" % CONFIG_PATH)
        except Exception as e:
            logger.error("保存配置失败: %s" % e)

    def _load_config(self):
        try:
            if CONFIG_PATH.exists():
                with open(CONFIG_PATH, "r", encoding="utf-8") as f:
                    cfg = json.load(f)
                if cfg.get("input_path"):
                    self.input_path.set(cfg["input_path"])
                if cfg.get("output_dir"):
                    self.output_dir.set(cfg["output_dir"])
                logger.info("已加载配置: %s" % CONFIG_PATH)
        except Exception as e:
            logger.error("加载配置失败: %s" % e)

    # ---------- 执行 ----------
    def _start(self):
        # 校验参数
        # ...

        # 保存配置
        self._save_config()

        # 业务逻辑
        logger.info("开始执行任务...")
        # ...


if __name__ == "__main__":
    root = tk.Tk()
    app = App(root)
    root.mainloop()
```

## 四、注意事项

| 项目 | 说明 |
|------|------|
| `SCRIPT_NAME` | 不含扩展名，如 `"file_mover"` 而非 `"file_mover.pyw"` |
| 配置键名 | 与 `StringVar` 变量名保持一致，方便维护 |
| `BooleanVar` 保存 | 直接 `json.dump` 即可，`True/False` 会序列化为 `true/false` |
| `sys.path` 插入 | 必须在 `from log_utils import` **之前**执行 |
| 导入失败 | `DummyLogger` 兜底，脚本照常运行，不影响任何业务逻辑 |
| 日志调用 | `logger.info()` / `logger.warning()` / `logger.error()` / `logger.debug()` |
