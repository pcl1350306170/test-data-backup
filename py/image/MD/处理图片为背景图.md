    * 或纯色背景 C（支持颜色选择器）

    * 尺寸缩放、位置（X/Y）、透明度

## 🧩 脚本代码（`image_composer.py`）

```python
import os
import json
import tkinter as tk
from tkinter import filedialog, colorchooser, messagebox
from PIL import Image, ImageTk, ImageEnhance

CONFIG_PATH = "image_config.json"

class ImageComposerApp:
    def __init__(self, root):
        self.root = root
        self.root.title("🐶 图片合成工具 - by GPT-5")
        self.root.geometry("1200x700")

        # 当前数据
        self.image_a = None
        self.bg_image = None
        self.result_image = None
        self.preview_imgtk = None

        # 参数
        self.params = {
            "scale": 100,
            "pos_x": 0,
            "pos_y": 0,
            "alpha": 100,
            "bg_mode": "color",
            "bg_color": "#FFFFFF",
            "output_format": "PNG",
            "output_width": 800,
            "output_height": 600,
            "jpg_quality": 90
        }

        self.load_config()
        self.build_ui()

    # ---------------- UI 构建 ----------------
    def build_ui(self):
        control_frame = tk.Frame(self.root)
        control_frame.pack(side="left", fill="y", padx=10, pady=10)

        preview_frame = tk.Frame(self.root, bg="#ddd")
        preview_frame.pack(side="right", fill="both", expand=True)

        # 控件部分
        tk.Label(control_frame, text="🎯 图片A").pack(anchor="w")
        tk.Button(control_frame, text="选择图片A", command=self.load_image_a).pack(fill="x")

        tk.Label(control_frame, text="🎨 背景设置").pack(anchor="w", pady=(10, 0))
        tk.Button(control_frame, text="选择背景图B", command=self.load_bg_image).pack(fill="x")
        tk.Button(control_frame, text="选择纯色背景C", command=self.choose_bg_color).pack(fill="x")

        tk.Label(control_frame, text="🔧 调节参数").pack(anchor="w", pady=(10, 0))
        self.scale_var = tk.DoubleVar(value=self.params["scale"])
        tk.Scale(control_frame, from_=10, to=300, orient="horizontal", label="缩放比例(%)", variable=self.scale_var, command=lambda _: self.update_preview()).pack(fill="x")

        self.alpha_var = tk.DoubleVar(value=self.params["alpha"])
        tk.Scale(control_frame, from_=0, to=100, orient="horizontal", label="透明度(%)", variable=self.alpha_var, command=lambda _: self.update_preview()).pack(fill="x")

        tk.Label(control_frame, text="位置X / Y").pack(anchor="w")
        pos_frame = tk.Frame(control_frame)
        pos_frame.pack(fill="x")
        self.pos_x_var = tk.IntVar(value=self.params["pos_x"])
        self.pos_y_var = tk.IntVar(value=self.params["pos_y"])
        tk.Entry(pos_frame, textvariable=self.pos_x_var, width=6).pack(side="left")
        tk.Entry(pos_frame, textvariable=self.pos_y_var, width=6).pack(side="left", padx=5)

        tk.Button(control_frame, text="居中显示", command=self.center_image).pack(fill="x")

        tk.Label(control_frame, text="💾 输出设置").pack(anchor="w", pady=(10, 0))
        self.format_var = tk.StringVar(value=self.params["output_format"])
        tk.OptionMenu(control_frame, self.format_var, "PNG", "JPG", "JPEG", "WebP").pack(fill="x")

        tk.Label(control_frame, text="JPG质量(0-100)").pack(anchor="w")
        self.quality_var = tk.IntVar(value=self.params["jpg_quality"])
        tk.Entry(control_frame, textvariable=self.quality_var).pack(fill="x")

        tk.Button(control_frame, text="预览", command=self.update_preview).pack(fill="x", pady=(10, 0))
        tk.Button(control_frame, text="保存输出", command=self.save_output).pack(fill="x", pady=(5, 0))

        tk.Button(control_frame, text="退出", command=self.on_exit).pack(fill="x", pady=(10, 0))

        # 预览区
        self.preview_canvas = tk.Canvas(preview_frame, bg="#eee")
        self.preview_canvas.pack(fill="both", expand=True)

    # ---------------- 功能逻辑 ----------------
    def load_image_a(self):
        path = filedialog.askopenfilename(title="选择图片A", filetypes=[("Images", "*.png;*.jpg;*.jpeg;*.gif;*.webp")])
        if not path: return
        self.image_a = Image.open(path).convert("RGBA")
        self.update_preview()

    def load_bg_image(self):
        path = filedialog.askopenfilename(title="选择背景图B", filetypes=[("Images", "*.png;*.jpg;*.jpeg;*.gif;*.webp")])
        if not path: return
        self.bg_image = Image.open(path).convert("RGBA")
        self.params["bg_mode"] = "image"
        self.update_preview()

    def choose_bg_color(self):
        color = colorchooser.askcolor(title="选择背景颜色")
        if color and color[1]:
            self.params["bg_mode"] = "color"
            self.params["bg_color"] = color[1]
            self.update_preview()

    def center_image(self):
        if self.bg_image:
            bg_w, bg_h = self.bg_image.size
        else:
            bg_w, bg_h = self.params["output_width"], self.params["output_height"]

        self.pos_x_var.set(bg_w // 2)
        self.pos_y_var.set(bg_h // 2)
        self.update_preview()

    def update_preview(self):
        if not self.image_a:
            return

        # 背景
        if self.params["bg_mode"] == "image" and self.bg_image:
            bg = self.bg_image.copy()
        else:
            bg = Image.new("RGBA", (self.params["output_width"], self.params["output_height"]), self.params["bg_color"])

        # 缩放与透明度
        scale = self.scale_var.get() / 100.0
        new_size = (int(self.image_a.width * scale), int(self.image_a.height * scale))
        img_a_resized = self.image_a.resize(new_size, Image.LANCZOS)

        alpha = self.alpha_var.get() / 100.0
        if alpha < 1.0:
            enhancer = ImageEnhance.Brightness(img_a_resized)
            img_a_resized = img_a_resized.convert("RGBA")
            img_a_resized.putalpha(int(255 * alpha))

        # 合成
        pos_x, pos_y = self.pos_x_var.get(), self.pos_y_var.get()
        bg.paste(img_a_resized, (pos_x, pos_y), img_a_resized)
        self.result_image = bg

        # 显示预览
        preview = bg.resize((600, 400), Image.LANCZOS)
        self.preview_imgtk = ImageTk.PhotoImage(preview)
        self.preview_canvas.delete("all")
        self.preview_canvas.create_image(300, 200, image=self.preview_imgtk)

    def save_output(self):
        if not self.result_image:
            messagebox.showwarning("提示", "请先生成预览")
            return
        path = filedialog.asksaveasfilename(defaultextension=f".{self.format_var.get().lower()}",
                                            filetypes=[("Image files", "*.png;*.jpg;*.jpeg;*.webp")])
        if not path: return

        fmt = self.format_var.get()
        params = {}
        if fmt.lower() in ["jpg", "jpeg"]:
            params["quality"] = self.quality_var.get()
            self.result_image = self.result_image.convert("RGB")

        self.result_image.save(path, fmt, **params)
        messagebox.showinfo("成功", f"文件已保存：{path}")

    def load_config(self):
        if os.path.exists(CONFIG_PATH):
            try:
                with open(CONFIG_PATH, "r", encoding="utf-8") as f:
                    self.params.update(json.load(f))
            except:
                pass

    def save_config(self):
        with open(CONFIG_PATH, "w", encoding="utf-8") as f:
            json.dump(self.params, f, ensure_ascii=False, indent=4)

    def on_exit(self):
        self.save_config()
        self.root.destroy()


if __name__ == "__main__":
    root = tk.Tk()
    app = ImageComposerApp(root)
    root.mainloop()
```
